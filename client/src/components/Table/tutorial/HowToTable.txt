// ==============================================
//      HOW THE TABLE CONTROLLER WORKS
// ==============================================


    In your component, import the <TableController /> which takes at the very least 3 props:
        variant         {string}    - type of table (basic, sortable, data, etc.)
        columns         {array}     - data columns (headers)
        data            {array}     - actual data to be displayed

        *optional props:
        onAction        {fn}        - (actionId, row) => void
        onRowClick      {fn}        - (row) => void
        onFetch         {fn}        - (params) => void (ServerTable only)
        caption         {string}    - accessible table caption
        emptyMessage    {string}    - shown when data is empty
        className       {string}    - extra class on the wrapper
        renderModal     {fn}        - (row) => <JSX> (in conjunction with onRowClick)

            isOpen: false,
            rowData: null
        });

        // modal funcs
        const openModal = (row) => {
            setModalState({isOpen: true, rowData: row});            
        }

        const closeModal = () => {
            setModalState({isOpen: false, rowData: null});
        }

        <div class='my basic table with modal popup'>
            <TableController
                variant='basic'
                columns={MY_COLS}
                data={MY_DATA}
                onRowClick={openModal}
            />
            <RowModal
                isOpen={modalState.isOpen}
                rowData={modalState.rowData}
                onClose={closeModal}
                title={'This is my Modal popup title!'}

                        <h3>{row.name}</h3>

                )}
            />
